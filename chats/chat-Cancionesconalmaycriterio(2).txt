```python
                "Pre-Coro": ["Dm9", "Em7", "Fmaj7", "G7alt"],
                "Coro": ["Am9", "Fmaj7", "Cmaj7", "E7#9"],
                "Verso 2": ["Am9", "Fmaj7", "Cmaj7", "G6"],
                "Puente": ["Fm", "Cm", "Db", "Eb"], 
                "Outro": ["Am9", "Fmaj7"]
            }},
            "vocal_direction": "Indicaciones de flow, respiraciones y actitud para Pedro en este idioma."
        }}
        """
        return prompt

    def compose(self, language, theme, bpm=105, key="Am"):
        """Ejecuta la composición completa."""
        print(f"🧠 ACB: Pensando nativamente en {language}...")
        
        style_matrix = {
            "theme": theme,
            "bpm": bpm,
            "key": key
        }
        
        system_prompt = self._build_native_system_prompt(language, style_matrix)
        
        # 1. LLAMADA AL CEREBRO LOCAL (OLLAMA)
        response = ollama.chat(model=self.model, messages=[
            {'role': 'system', 'content': system_prompt},
            {'role': 'user', 'content': f"Genera la obra maestra para Pedro."}
        ], format='json') # Fuerza a la IA a devolver JSON limpio
        
        song_data = json.loads(response['message']['content'])
        
        # 2. MOTOR DE TEORÍA MUSICAL (Generación de MIDI)
        print("🎼 ACB: Traduciendo armonía a MIDI...")
        self.midi.addTempo(self.track, self.time, bpm)
        
        current_time = 0
        for section in song_data['structure']:
            if section in song_data['chords']:
                chords = song_data['chords'][section]
                for chord in chords:
                    # Limpiar el nombre del acorde para buscarlo en el mapa
                    clean_chord = chord.replace('(', '').replace(')', '').split('/')[0]
                    notes = self.chord_map.get(clean_chord, [60, 64, 67]) # Fallback a Do Mayor
                    
                    for note in notes:
                        # Añade cada nota del acorde, duración 4 beats
                        self.midi.addNote(self.track, 0, note, current_time, 4, 100)
                    current_time += 4
                    
        # Guardar MIDI
        midi_filename = f"Belentani_{song_data['title'].replace(' ', '_')}.mid"
        with open(midi_filename, 'wb') as f:
            self.midi.writeFile(f)
            
        # 3. GUARDAR LETRA Y ESTRUCTURA
        txt_filename = f"Belentani_{song_data['title'].replace(' ', '_')}_Lyrics.txt"
        with open(txt_filename, 'w', encoding='utf-8') as f:
            f.write(f"TÍTULO: {song_data['title']}\n")
            f.write(f"IDIOMA: {language.upper()} | BPM: {bpm} | TONALIDAD: {key}\n")
            f.write(f"DIRECCIÓN VOCAL: {song_data['vocal_direction']}\n")
            f.write("="*50 + "\n\n")
            for section in song_data['structure']:
                f.write(f"[{section}]\n")
                f.write(f"Acordes: {' | '.join(song_data['chords'].get(section, []))}\n")
                f.write(f"{song_data['lyrics'].get(section, '(Instrumental)')}\n\n")

        print(f"✅ OBRA MAESTRA GENERADA:")
        print(f"📄 Letra y Acordes: {txt_filename}")
        print(f"🎹 Archivo MIDI (para tu DAW): {midi_filename}")
        return song_data

# ==========================================
# EJECUCIÓN DEL ALGORITMO
# ==========================================
if __name__ == "__main__":
    # Inicializa el motor (asegúrate de tener Ollama corriendo con 'llama3' o 'qwen2.5')
    engine = BelentaniEngine(model_name="llama3") 
    
    # PARAMETROS DE ENTRADA PARA PEDRO BELENTANI
    idioma_deseado = "frances" # Puedes poner: portugues, espanol, ingles, catalan, frances
    tema = "La hipocresía de una fiesta de alta sociedad donde todos fingen ser felices"
    tempo = 110
    tonalidad = "Cm"
    
    engine.compose(
        language=idioma_deseado, 
        theme=tema, 
        bpm=tempo, 
        key=tonalidad
    )
```

### 💻 CÓDIGO v2.0: Motor Belentani Blindado (Python)

Este código anticipa los fallos, limpia la basura que escupe la IA y te da el resultado aunque la IA se comporte mal.

```python
import ollama
import json
import re
import sys
from midiutil import MIDIFile

class BelentaniEngineV2:
    def __init__(self, model_name="qwen2.5:7b"): 
        # CAMBIO CRÍTICO 1: Uso Qwen 2.5 7B. Es mucho mejor en español/catalán/francés 
        # que Llama 3, y el tamaño 7B corre en casi cualquier PC con 8GB RAM.
        self.model = model_name
        self._check_ollama_connection()
        
        # Diccionario de acordes EXPANDIDO y con fallback
        self.chord_map = self._build_robust_chord_map()

    def _check_ollama_connection(self):
        """Verifica si Ollama está corriendo antes de intentar nada."""
        try:
            ollama.list()
            print("✅ Conexión con Ollama establecida.")
        except Exception as e:
            print("❌ ERROR FATAL: Ollama no está corriendo.")
            print("👉 Solución: Abre la aplicación Ollama en tu ordenador o ejecuta 'ollama serve' en la terminal.")
            sys.exit(1)

    def _build_robust_chord_map(self):
        """Mapeo de notas MIDI con variaciones comunes para evitar errores de KeyError."""
        base_map = {
            'C': [60, 64, 67], 'Cm': [60, 63, 67], 'Cmaj7': [60, 64, 67, 71], 
            'Cm7': [60, 63, 67, 70], 'Cm9': [60, 63, 67, 70, 74],
            'Dm': [62, 65, 69], 'Dm7': [62, 65, 69, 72], 'Dm9': [62, 65, 69, 72, 76],
            'Em': [64, 67, 71], 'Em7': [64, 67, 71, 74],
            'F': [65, 69, 72], 'Fmaj7': [65, 69, 72, 76], 'Fm': [65, 68, 72],
            'G': [67, 71, 74], 'G7': [67, 71, 74, 77], 'Gmaj7': [67, 71, 74, 78],
            'Am': [69, 72, 76], 'Am7': [69, 72, 76, 79], 'Am9': [69, 72, 76, 79, 83],
            'Bb': [70, 74, 77], 'Bbmaj7': [70, 74, 77, 81],
            'Eb': [63, 67, 70], 'Ebmaj7': [63, 67, 70, 74],
            'Ab': [68, 72, 75], 'Db': [61, 65, 68]
        }
        return base_map

    def _clean_json_from_llm(self, raw_text):
        """
        CAMBIO CRÍTICO 2: Las IAs adoran poner ```json al principio y ``` al final.
        Esta función lo limpia para que json.loads() no explote.
        """
        # Busca el primer '{' y el último '}'
        start = raw_text.find('{')
        end = raw_text.rfind('}')
        if start != -1 and end != -1:
            return raw_text[start:end+1]
        return raw_text

    def _parse_chord_safely(self, chord_str, fallback_key="Am"):
        """
        CAMBIO CRÍTICO 3: Si la IA inventa "Cmaj13#11", el código no crashea.
        Busca la raíz y devuelve un acorde seguro.
        """
        # Extrae solo la raíz y el tipo básico (ej: "Cmaj7" de "Cmaj7/F")
        clean_chord = chord_str.split('/')[0].strip()
        
        # Busca coincidencia exacta
        if clean_chord in self.chord_map:
            return self.chord_map[clean_chord]
        
        # Busca coincidencia parcial (ej: si dice "Cmaj9", usa "Cmaj7")
        for key in self.chord_map:
            if key in clean_chord:
                return self.chord_map[key]
                
        # FALLBACK: Si inventa algo raro, devuelve la tónica de la tonalidad pedida
        print(f"⚠️ Acorde inventado por IA ('{chord_str}'). Usando fallback.")
        return self.chord_map.get(fallback_key, [60, 64, 67])

    def _build_mega_prompt(self, language, theme, bpm, key):
        """Prompt de nivel Dios. Obliga a la IA a pensar en fonética y estructura."""
        
        phonetic_directives = {
            "portugues": "REGLA DE ORO: Pense em português brasileiro nativo. Use muitas vogais nasais (ão, em, am) no refrão para sustentar notas longas. Consoantes oclusivas (p, t, k, b) no verso para criar percussão vocal (estilo Anitta).",
            "espanol": "REGLA DE ORO: Piensa nativamente en español rioplatense/urbano. Usa fricativas y vibrantes (rr, ll, y) para actitud y tensión (estilo Nathy Peluso). Juega con el ritmo de las sílabas, no solo con la melodía.",
            "ingles": "GOLDEN RULE: Think natively in English. Use stress-timed rhythm. Pack consonant clusters (tr, bl, str, mp) in the verses for percussive flow. Open vowels (a, o) on the high belts in the chorus (Gaga style).",
            "catalan": "REGLA D'OR: Pensa nativament en català. Explota les vocals neutres i tancades per a la intimitat dels versos. Obre les vocals (a, e, o) en el tornada. Usa la 'x' i 'ss' per donar textura sibilant i fosca.",
            "frances": "RÈGLE D'OR: Pense nativement en français. Utilise les liaisons pour lier les mots. Exploite les voyelles nasales (an, in, on, un) pour la mélancolía et l'élégance. Le rythme doit être legato mais avec un flow moderne."
        }

        return f"""
        Eres el ALGORITMO DE COMPOSICIÓN BELENTANI (ACB) v2.0.
        {phonetic_directives.get(language, phonetic_directives['espanol'])}
        
        CONTEXTO ARTÍSTICO (PEDRO BELENTANI):
        Fusión de: Teatralidad Pop (Gaga) + Melancolía Indie (The Smiths) + Groove Percusivo (Anitta) + Tensión Armónica (Nathy Peluso).
        
        PARÁMETROS:
        - Idioma: {language.upper()}
        - Tema: {theme}
        - BPM: {bpm}
        - Tonalidad: {key}
        
        INSTRUCCIONES TÉCNICAS:
        1. Los acordes deben ser complejos (usa 7, 9, maj7, m7b5, alt). Nada de tríadas básicas.
        2. La estructura debe tener dinámicas extremas (versos minimalistas, estribillos densos).
        3. Añade un campo "syllable_stress" en el JSON que marque con MAYÚSCULAS la sílaba fuerte de cada verso para guiar el flow rítmico.

        DEVUELVE ÚNICAMENTE UN JSON VÁLIDO (SIN MARKDOWN, SIN TEXTO EXTRA) CON ESTA ESTRUCTURA:
        {{
            "title": "Título",
            "bpm": {bpm},
            "key": "{key}",
            "structure": ["Verso 1", "Pre-Coro", "Coro", "Verso 2", "Puente", "Coro", "Outro"],
            "lyrics": {{
                "Verso 1": "lEtra nAtIva cOn sílAbas fUErtes en mAyÚscuLAs...",
                "Pre-Coro": "...",
                "Coro": "...",
                "Verso 2": "...",
                "Puente": "...",
                "Outro": "..."
            }},
            "chords": {{
                "Verso 1": ["Am9", "Fmaj7", "Cmaj7", "E7#9"],
                "Pre-Coro": ["Dm9", "Em7b5", "A7alt", "Dm9"],
                "Coro": ["Fmaj7", "Cmaj7", "Am7", "G7sus4"],
                "Verso 2": ["Am9", "Fmaj7", "Cmaj7", "E7#9"],
                "Puente": ["Fm", "Cm", "Dbmaj7", "Bb7"],
                "Outro": ["Am9", "Fmaj7"]
            }},
            "vocal_direction": "Instrucciones técnicas de respiración, tono y actitud para Pedro."
        }}
        """

    def compose(self, language, theme, bpm=105, key="Am"):
        print(f"\n🧠 [ACB v2.0] Iniciando composición nativa en {language}...")
        print(f"🎼 Tema: {theme} | BPM: {bpm} | Tonalidad: {key}\n")
        
        prompt = self._build_mega_prompt(language, theme, bpm, key)
        
        # 1. LLAMADA A LA IA CON REINTENTOS
        try:
            response = ollama.chat(
                model=self.model, 
                messages=[
                    {'role': 'system', 'content': 'Eres un motor de composición de élite. Devuelve SOLO JSON válido.'},
                    {'role': 'user', 'content': prompt}
                ],
                format='json', # Fuerza formato JSON en Ollama
                options={'temperature': 0.8} # Un poco de creatividad, pero controlada
            )
            raw_content = response['message']['content']
        except Exception as e:
            print(f"❌ Error al llamar a Ollama: {e}")
            return None

        # 2. LIMPIEZA Y PARSEO DEL JSON (El escudo anti-alucinación)
        try:
            clean_json = self._clean_json_from_llm(raw_content)
            song_data = json.loads(clean_json)
            print("✅ JSON de la IA parseado correctamente.")
        except json.JSONDecodeError as e:
            print(f"❌ La IA devolvió JSON inválido. Error: {e}")
            print("👉 Respuesta cruda de la IA:\n", raw_content[:500])
            return None

        # 3. GENERACIÓN MIDI CON FALLBACKS
        print("🎹 Traduciendo armonía a MIDI (con protección anti-acordes raros)...")
        midi = MIDIFile(1)
        track = 0
        time = 0
        midi.addTempo(track, time, song_data.get('bpm', bpm))
        
        for section in song_data.get('structure', []):
            section_chords = song_data.get('chords', {}).get(section, [])
            for chord_str in section_chords:
                notes = self._parse_chord_safely(chord_str, key)
                for note in notes:
                    midi.addNote(track, 0, note, time, 4, 100)
                time += 4
                
        safe_title = re.sub(r'[^\w\-_. ]', '_', song_data.get('title', 'Untitled'))
        midi_filename = f"Belentani_{safe_title}.mid"
        with open(midi_filename, 'wb') as f:
            midi.writeFile(f)
            
        # 4. EXPORTAR LETRA CON GUÍA RÍTMICA
        txt_filename = f"Belentani_{safe_title}_Lyrics.txt"
        with open(txt_filename, 'w', encoding='utf-8') as f:
            f.write(f"🎵 TÍTULO: {song_data.get('title')}\n")
            f.write(f"🌍 IDIOMA: {language.upper()} | ⏱️ BPM: {song_data.get('bpm')} | 🎹 TONALIDAD: {song_data.get('key')}\n")
            f.write(f"🗣️ DIRECCIÓN VOCAL: {song_data.get('vocal_direction')}\n")
            f.write("="*60 + "\n\n")
            for section in song_data.get('structure', []):
                f.write(f"--- [{section.upper()}] ---\n")
                chords = song_data.get('chords', {}).get(section, [])
                f.write(f"Acordes: {' | '.join(chords)}\n")
                f.write(f"Letra: {song_data.get('lyrics', {}).get(section, '(Instrumental)')}\n\n")

        print(f"\n🎉 ¡OBRA MAESTRA GENERADA Y BLINDADA!")
        print(f"📄 Letra y Guía Rítmica: {txt_filename}")
        print(f"🎹 Archivo MIDI (Ábrela en tu DAW): {midi_filename}")
        return song_data

# ==========================================
# EJECUCIÓN
# ==========================================
if __name__ == "__main__":
    # Asegúrate de haber descargado el modelo: ollama pull qwen2.5:7b
    engine = BelentaniEngineV2(model_name="qwen2.5:7b") 
    
    engine.compose(
        language="catalan", 
        theme="La soledad de ser el centro de atención en una gala benéfica, rodeado de gente falsa", 
        bpm=95, 
        key="Dm"
    )
```

### 🛡️ MANUAL DE ERRORES Y CORRECCIONES (Lo que te iba a pasar)

Si has intentado hacer esto antes, aquí tienes la explicación de por qué falló y cómo este código v2.0 lo soluciona:

#### Error 1: `json.decoder.JSONDecodeError: Expecting value: line 1 column 1`
*   **Por qué pasaba:** Le pediste a la IA un JSON. La IA te respondió: `Aquí tienes tu canción: \n ```json \n {"title":...} \n ``` `. Python no sabe leer markdown, solo JSON puro.
*   **La Corrección v2.0:** Añadí la función `_clean_json_from_llm()`. Busca el primer `{` y el último `}` y borra toda la basura de alrededor. Además, usamos `format='json'` en la llamada a Ollama, que fuerza al modelo a no escupir texto extra.

#### Error 2: `KeyError: 'Fmaj9'` (El script se rompe al generar el MIDI)
*   **Por qué pasaba:** Tu diccionario de acordes solo tenía `Fmaj7`. La IA, siendo "creativa", generó un `Fmaj9`. Python buscó `Fmaj9` en el diccionario, no lo encontró, y el programa murió.
*   **La Corrección v2.0:** Añadí `_parse_chord_safely()`. Si la IA inventa un acorde, el código busca la raíz (`F`) y el tipo más cercano (`maj7`) y lo usa. Si todo falla, usa la tónica de la tonalidad. **El script nunca se romperá por un acorde.**

#### Error 3: `ollama._types.ResponseError: model "llama3" not found`
*   **Por qué pasaba:** Copiaste el código pero no descargaste el modelo en tu terminal, o escribiste mal el nombre.
*   **La Corrección v2.0:** 
    1. Cambié el modelo por defecto a `qwen2.5:7b`. **Qwen es infinitamente superior a Llama 3 para español, catalán y francés**, y entiende mejor las instrucciones complejas de fonética.
    2. Añadí `_check_ollama_connection()` para que te diga exactamente qué hacer si Ollama no está abierto.

#### Error 4: La IA traduce literalmente y suena robótica.
*   **Por qué pasaba:** Si le dices "Escribe en francés sobre la soledad", la IA piensa en español, traduce palabra por palabra, y el francés suena a Google Translate.
*   **La Corrección v2.0:** El prompt ahora incluye **Reglas de Oro Fonéticas**. No le pide que "escriba en francés", le dice: *"Pense nativement en français. Utilise les liaisons... Exploite les voyelles nasales"*. Obliga a la IA a acceder a su red neuronal en ese idioma específico, priorizando la *boca y el oído* sobre el diccionario.

#### Error 5: El Flow rítmico es plano.
*   **Por qué pasaba:** La IA escribe poesía, no letras de canción. No sabe dónde cae el golpe fuerte del bombo.
*   **La Corrección v2.0:** Añadí el parámetro `"syllable_stress"`. Le obligo a la IA a escribir la letra poniendo en **MAYÚSCULAS** la sílaba que debe acentuar rítmicamente. 
    *   *Ejemplo de salida:* "lA sOledAd en El cEntrO de la gAla". 
    *   Esto te da una guía visual exacta de dónde poner el *groove* al cantarlo.

```python
import streamlit as st
import ollama
import json
import re

from midiutil import MIDIFile

st.set_page_config(page_title="Cockpit Belentani v3.0", page_icon="🎹", layout="wide")

def clean_json_from_llm(raw_text):
    start = raw_text.find('{')
    end = raw_text.rfind('}')
    if start != -1 and end != -1:
        return raw_text[start:end+1]
    return raw_text

def parse_chord_safely(chord_str, fallback_key="Am"):
    chord_map = {
        'C': [60, 64, 67], 'Cm': [60, 63, 67], 'Cmaj7': [60, 64, 67, 71], 'Cm7': [60, 63, 67, 70], 'Cm9': [60, 63, 67, 70, 74],
        'Dm': [62, 65, 69], 'Dm7': [62, 65, 69, 72], 'Dm9': [62, 65, 69, 72, 76],
        'Em': [64, 67, 71], 'Em7': [64, 67, 71, 74], 'F': [65, 69, 72], 'Fmaj7': [65, 69, 72, 76], 'Fm': [65, 68, 72],
        'G': [67, 71, 74], 'G7': [67, 71, 74, 77], 'Gmaj7': [67, 71, 74, 78],
        'Am': [69, 72, 76], 'Am7': [69, 72, 76, 79], 'Am9': [69, 72, 76, 79, 83],
        'Bb': [70, 74, 77], 'Bbmaj7': [70, 74, 77, 81], 'Eb': [63, 67, 70], 'Ebmaj7': [63, 67, 70, 74]
    }
    clean_chord = chord_str.split('/')[0].strip()
    if clean_chord in chord_map: return chord_map[clean_chord]
    for key in chord_map:
        if key in clean_chord: return chord_map[key]
    return chord_map.get(fallback_key, [60, 64, 67])

def build_mega_prompt(language, styles, theme, bpm, key):
    style_matrix = {
        "Lady Gaga (Teatralidad Pop)": "Estructuras dinámicas extremas, puentes que rompen la métrica, vulnerabilidad emocional.",
        "The Smiths (Melancolía Indie)": "Intercambio modal (acordes menores paralelos), letras con ironía trágica, guitarras jangly.",
        "Anitta (Groove Percusivo)": "Síncopas marcadas, consonantes oclusivas (p,t,k) como percusión, flow de favela funk.",
        "Nathy Peluso (Tensión Armónica)": "Acordes de jazz/tango (m7b5, alt), rubato, disonancias resueltas, actitud visceral."
    }
    
    selected_styles_text = " + ".join([style_matrix[s] for s in styles])
    
    phonetic_directives = {
        "Portugués": "Pense em português brasileiro nativo. Vogais nasais no refrão, consoantes oclusivas no verso.",
        "Español": "Piensa nativamente en español. Fricativas y vibrantes para actitud, juego rítmico de sílabas.",
        "Inglés": "Think natively in English. Stress-timed rhythm, consonant clusters for percussive flow.",
        "Catalán": "Pensa nativament en català. Contrast de vocals obertes/tancades, textura sibilant.",
        "Francés": "Pense nativement en français. Liaisons, voyelles nasales pour la mélancolie, legato moderne."
    }

    return f"""
    Eres el ALGORITMO DE COMPOSICIÓN BELENTANI.
    {phonetic_directives.get(language, phonetic_directives['Español'])}
    
    ESTILO MUSICAL (Fusión): {selected_styles_text}
    TEMA: {theme}
    BPM: {bpm} | TONALIDAD: {key}
    
    INSTRUCCIONES:
    1. Acordes complejos (7, 9, maj7, m7b5, alt).
    2. Estructura con dinámicas extremas.
    3. Pon en MAYÚSCULAS la sílaba fuerte de cada verso para guiar el flow.

    DEVUELVE SOLO JSON VÁLIDO:
    {{
        "title": "Título",
        "bpm": {bpm},
        "key": "{key}",
        "structure": ["Verso 1", "Pre-Coro", "Coro", "Verso 2", "Puente", "Coro", "Outro"],
        "lyrics": {{"Verso 1": "...", "Pre-Coro": "...", "Coro": "...", "Verso 2": "...", "Puente": "...", "Outro": "..."}},
        "chords": {{"Verso 1": ["Am9", "Fmaj7"], "Pre-Coro": ["Dm9", "E7alt"], "Coro": ["Fmaj7", "Cmaj7"], "Verso 2": ["Am9", "Fmaj7"], "Puente": ["Fm", "Cm"], "Outro": ["Am9"]}},
        "vocal_direction": "Instrucciones de actitud y respiración."
    }}
    """

def generate_midi(song_data):
    midi = MIDIFile(1)
    track = 0
    time = 0
    midi.addTempo(track, time, song_data.get('bpm', 100))
    for section in song_data.get('structure', []):
        for chord_str in song_data.get('chords', {}).get(section, []):
            notes = parse_chord_safely(chord_str, song_data.get('key', 'Am'))
            for note in notes:
                midi.addNote(track, 0, note, time, 4, 100)
            time += 4
    return midi

with col2:

theme = st.text_area("¿Qué vamos a componer hoy? (Sé tan específico, raro o poético como quieras)", 
                     placeholder="Ej: La hipocresía de una fiesta de alta sociedad donde todos fingen ser felices, pero el protagonista solo quiere escapar...")

# BOTÓN DE GENERACIÓN
if st.button("🚀 GENERAR OBRA MAESTRA", type="primary", use_container_width=True):
    if not theme:
        st.warning("Por favor, introduce un concepto o tema.")
    elif not styles:
        st.warning("Por favor, elige al menos un estilo musical.")
    else:
        with st.spinner("🧠 El Algoritmo está pensando nativamente en el idioma y fusionando estilos..."):
            try:
                prompt = build_mega_prompt(language, styles, theme, bpm, key)
                response = ollama.chat(
                    model=model, 
                    messages=[
                        {'role': 'system', 'content': 'Eres un motor de composición. Devuelve SOLO JSON válido.'},
                        {'role': 'user', 'content': prompt}
                    ],
                    format='json',
                    options={'temperature': 0.8}
                )
                
                clean_json = clean_json_from_llm(response['message']['content'])
                song_data = json.loads(clean_json)
                
                st.success("✅ ¡Obra Maestra Generada!")
                
                # Mostrar Letra y Acordes
                st.subheader(f"🎵 {song_data.get('title', 'Sin Título')}")
                st.caption(f"🗣️ Dirección Vocal: {song_data.get('vocal_direction', '')}")
                
                for section in song_data.get('structure', []):
                    with st.expander(f"🎼 [{section.upper()}]"):
                        chords = song_data.get('chords', {}).get(section, [])
                        lyrics = song_data.get('lyrics', {}).get(section, "(Instrumental)")
                        st.markdown(f"**Acordes:** `{' | '.join(chords)}`")
                        st.markdown(f"**Letra:**\n{lyrics}")
                
                # Botones de Descarga
                col_dl1, col_dl2 = st.columns(2)
                
                # Descargar TXT
                txt_content = f"TÍTULO: {song_data.get('title')}\nIDIOMA: {language} | BPM: {bpm} | TONALIDAD: {key}\n\n"
                for section in song_data.get('structure', []):
                    txt_content += f"[{section}]\nAcordes: {' | '.join(song_data.get('chords', {}).get(section, []))}\n{song_data.get('lyrics', {}).get(section, '')}\n\n"
                
                with col_dl1:
                    st.download_button(
                        label="📄 Descargar Letra y Acordes (TXT)",
                        data=txt_content,
                        file_name=f"Belentani_{song_data.get('title', 'Track')}.txt",
                        mime="text/plain"
                    )
                    
                # Descargar MIDI
                midi_obj = generate_midi(song_data)
                midi_buffer = io.BytesIO()
                midi_obj.writeFile(midi_buffer)
                midi_buffer.seek(0)
                
                with col_dl2:
                    st.download_button(
                        label="🎹 Descargar Pista MIDI (Para tu DAW)",
                        data=midi_buffer,
                        file_name=f"Belentani_{song_data.get('title', 'Track')}.mid",
                        mime="audio/midi"
                    )

            except json.JSONDecodeError:
                st.error("❌ La IA ha escupido un JSON roto. Vuelve a darle al botón, a veces se atascan.")
            except Exception as e:
                st.error(f"❌ Error inesperado: {e}")
```

```python
                col_dl1, col_dl2 = st.columns(2)
                
                # TXT
                txt_data = f"TITULO: {cancion.get('titulo')}\n\n"
                for sec in cancion.get('estructura', []):
                    txt_data += f"[{sec}]\nAcordes: {' | '.join(cancion.get('acordes', {}).get(sec, []))}\n{cancion.get('letras', {}).get(sec, '')}\n\n"
                
                with col_dl1:
                    st.download_button("📄 Descargar Letra (TXT)", txt_data, file_name="letra.txt")
                
                # MIDI
                midi = MIDIFile(1)
                track = 0; time = 0
                midi.addTempo(track, time, bpm)
                for sec in cancion.get('estructura', []):
                    for acorde in cancion.get('acordes', {}).get(sec, []):
                        notas = acorde_a_midi(acorde, tonalidad)
                        for nota in notas:
                            midi.addNote(track, 0, nota, time, 4, 100)
                        time += 4
                
                buffer = io.BytesIO()
                midi.writeFile(buffer)
                buffer.seek(0)
                
                with col_dl2:
                    st.download_button("🎹 Descargar MIDI", buffer, file_name="musica.mid", mime="audio/midi")

            except Exception as e:
                st.error(f"❌ Error en el Agente: {e}")
                st.info("Asegúrate de que Ollama está corriendo y el modelo descargado.")
```

        midi = MIDIFile(1)
        track = 0
        time = 0
        midi.addTempo(track, time, song_data.get('bpm', 105))
        
        for section in song_data.get('structure', []):
            section_chords = song_data.get('chords', {}).get(section, [])
            for chord_str in section_chords:
                notes = self.parse_chord_safely(chord_str, song_data.get('key', 'Am'))
                for note in notes:
                    midi.addNote(track, 0, note, time, 4, 100)
                time += 4
        return midi

```python
    track_chords, track_bass, track_click = 0, 1, 2
    time = 0
    channel = 0
    
    for t in [track_chords, track_bass, track_click]:
        midi.addTempo(t, time, bpm)
        midi.addProgramChange(t, channel, time, 0 if t==0 else 33 if t==1 else 115) # Piano / Fingered Bass / Woodblock

        for acorde_str in acordes:
```

1. **Inmune a los "Fantasmas" de Streamlit:** Gracias a `st.session_state`, una vez que generas la canción, puedes bajar a la barra lateral, cambiar el BPM a 120 y el idioma a Catalán para preparar tu *siguiente* canción, y **la canción actual seguirá en pantalla intacta** hasta que le des a "Generar" de nuevo o pulses "Limpiar".

### 🚨 El Fallo Matriz: La IA es "Ciega y Sorda"
Los modelos como Qwen o Llama **no escuchan, no tienen boca y no entienden de fonemas**. Son modelos estadísticos de texto (tokens). 
Cuando le pides "escribe en catalán", la IA no está "sintiendo" la sonoridad de la lengua; simplemente está buscando en su base de datos qué palabras en catalán suelen ir juntas. Por eso genera letras que *gramaticalmente* son correctas, pero que al cantarlas **suenan a plástico, sin la "ginga" del portugués, sin la "sibilancia" del catalán, y con acentos musicales que chocan contra los acentos naturales de las palabras** (el error de prosodia).

Para que el Agente entienda la **esencia fonética y el ritmo real**, no podemos confiar solo en el LLM. Necesitamos inyectar en el código un **Auditor Fonético Externo**, un módulo de software que SÍ conozca las reglas físicas de cada idioma.

### 🛠️ La Solución Real: Arquitectura de 2 Fases (El Módulo Prosódico)

Para solucionarlo en tu ordenador, el Agente Omega debe dejar de trabajar en 1 paso y pasar a trabajar en **3 pasos de auditoría estricta**. Necesitamos integrar en tu script de Python una librería de **Fonética Computacional** (como `espeak-ng` o librerías de NLP específicas por idioma) que actúe como un "Profesor de Canto" que regaña a la IA.

El nuevo flujo de trabajo del Agente sería este:

#### FASE 1: El Borrador (La IA a ciegas)
La IA genera la letra basándose en el tema y los estilos. Esto es solo arcilla en bruto.

#### FASE 2: El Auditor Fonético (El Código Real)
El script de Python toma ese texto y lo pasa por un **Silabificador y Analizador de Acentos** que sí entiende la física del idioma.
*   **En Catalán:** El código detecta las vocales neutres, aplica las reglas de la *sinalefa* catalana y cuenta las sílabas métricas reales.
*   **En Portugués:** El código identifica obligatoriamente las vocales nasales (`ão`, `em`) y verifica que no estén colocadas en notas rápidas donde es imposible pronunciarlas.
*   **En Español:** Marca la sílaba tónica de cada palabra según las reglas de la RAE.

#### FASE 3: El Choque Prosódico (La Corrección Forzosa)
El código cruza los datos fonéticos con el BPM y el compás musical. Si detecta que la IA ha puesto una palabra grave (ej: *árbol*) en el golpe fuerte del bombo, o que ha puesto una vocal cerrada (`i`, `u`) en la nota más aguda del estribillo (donde un cantante se ahogaría), **el código rechaza la letra y le da un "Prompt de Regaño Técnico" a la IA** para que reescriba SOLO ese verso.

Código para un Prompt Builder fonético y prosódico útil por idioma
Te entrego un módulo Python listo para integrar en tu agente. Objetivo: transformar una instrucción genérica “hazlo en X idioma” en un prompt que aporte rasgos fonéticos, prosódicos y métricos concretos (sílabas tónicas, marcas de acento, transcripción fonética opcional) para que el modelo no haga solo una traducción literal.

Dependencias opcionales: phonemizer (para transcripción fonética), epitran (si quieres IPA más precisa), unidecode. El módulo funciona sin ellas (cae a modo heurístico).

Diseño: funciones para (1) normalizar texto, (2) marcar sílabas tónicas con MAYÚSCULAS o | para compases, (3) generar directivas fonéticas por idioma, (4) construir prompt final listo para enviar al LLM.

# phonetic_prompt_builder.py
import re
import unicodedata

# Optional libs: phonemizer or epitran for IPA. Use heuristics if not installed.
try:
    from phonemizer import phonemize
    HAS_PHONEMIZER = True
except Exception:
    HAS_PHONEMIZER = False

# -------------------------
# 1. Directivas fonéticas por idioma
# -------------------------
PHONETIC_DIRECTIVES = {
    "es": {
        "desc": "Piensa y escribe en español nativo. Usa vocales abiertas (a e o) para belting; marca vibración de r final; usa síncopa en versos.",
        "rules": ["marcar_tonica_mayusculas", "mapear_compases_4_4"]
    },
    "ca": {
        "desc": "Pensa i escriu en català. Allarga vocals obertes en tornada; marca síl·labes tònica en MAJÚSCULES; utilitza r vibrada lleu.",
        "rules": ["marcar_tonica_mayusculas", "mapear_compases_8_silabas"]
    },
    "pt": {
        "desc": "Pense em português. Marque nasalização con ~; sílabas tônicas en MAIÚSCULAS; ritmo bossa/groove.",
        "rules": ["marcar_tonica_mayusculas", "marcar_nasalizacion"]
    },
    "en": {
        "desc": "Think natively in English. Stress-timed rhythm; mark stressed syllables in CAPS; use consonant clusters as percussive elements.",
        "rules": ["marcar_tonica_mayusculas", "mapear_compases_4_4"]
    }
}

# -------------------------
# 2. Utilidades de texto y sílabas (heurísticas)
# -------------------------
VOWELS = "aeiouáéíóúàèìòùãõâêîôûäëïöü"

def strip_accents(text):
    return ''.join(c for c in unicodedata.normalize('NFD', text)
                   if unicodedata.category(c) != 'Mn')

def split_syllables_heuristic(word):
    """
    Heurística simple para dividir sílabas. No es perfecta, pero útil como fallback.
    Divide por vocales y diptongos básicos.
    """
    word = word.lower()
    # separar por consonante-vocal patrones simples
    parts = re.findall(r'[^aeiouáéíóúàèìòùãõâêîôûäëïöü]*[aeiouáéíóúàèìòùãõâêîôûäëïöü]+(?:n?g?l?r?)*', word, re.IGNORECASE)
    if parts:
        return parts
    # fallback: cada vocal es sílaba
    return re.findall(r'[aeiouáéíóúàèìòùãõâêîôûäëïöü]+|[^aeiou]+', word, re.IGNORECASE)

def mark_stress_in_line(line, language):
    """
    Marca la sílaba tónica en MAYÚSCULAS en cada palabra según heurística:
    - Si la palabra tiene tilde, esa sílaba.
    - Si no, aplica regla general simplificada (es/es: penúltima si termina en vocal/n/s).
    """
    def mark_word(w):
        if re.search(r'[áéíóúàèìòùãõâêîôû]', w, re.IGNORECASE):
            # encontrar sílaba con tilde
            parts = re.split(r'([aeiouáéíóúàèìòùãõâêîôûäëïöü]+)', w, flags=re.IGNORECASE)
            # reconstruir y poner mayúsculas en la sílaba con tilde
            for i, p in enumerate(parts):
                if re.search(r'[áéíóúàèìòùãõâêîôû]', p, re.IGNORECASE):
                    parts[i] = p.upper()
                    return ''.join(parts)
            return w
        # heurística por idioma
        syls = split_syllables_heuristic(w)
        if len(syls) == 1:
            return w.upper() if len(w) <= 3 else w  # monosílabos cortos: opcional
        # regla simplificada para español/catalán/portugués
        if language in ('es', 'ca', 'pt'):
            if re.search(r'[aeiouáéíóúàèìòùãõâêîôû]$', w, re.IGNORECASE) or re.search(r'[ns]$', w, re.IGNORECASE):
                # penúltima sílaba tónica
                idx = max(0, len(syls)-2)
            else:
                idx = len(syls)-1
        else:
            # inglés: stress often on first syllable (very rough)
            idx = 0
        # uppercase that syllable
        syls[idx] = syls[idx].upper()
        return ''.join(syls)

    # procesar cada palabra manteniendo puntuación
    tokens = re.findall(r"\w+['’]?\w*|[^\w\s]", line, re.UNICODE)
    rebuilt = []
    for t in tokens:
        if re.match(r'\w', t, re.UNICODE):
            rebuilt.append(mark_word(t))
        else:
            rebuilt.append(t)
    return ' '.join(rebuilt)

# -------------------------
# 3. IPA / fonética (opcional)
# -------------------------
def to_ipa(text, lang):
    """
    Intenta usar phonemizer si está disponible; si no, devuelve None.
    """
    if not HAS_PHONEMIZER:
        return None
    try:
        ipa = phonemize(text, language=lang, backend='espeak', strip=True, preserve_punctuation=True, with_stress=True)
        return ipa
    except Exception:
        return None

# -------------------------
# 4. Constructor de prompt fonético
# -------------------------
def build_phonetic_prompt(language_code, theme, style_matrix, bpm, key, seed_lines=None, ipa_enabled=False):
    """
    Devuelve un prompt robusto que incluye:
    - Directiva fonética por idioma
    - Reglas métricas y de prosodia
    - Letras de ejemplo marcadas con sílabas tónicas
    - (Opcional) IPA para líneas clave
    """
    lang = language_code.lower()
    directive = PHONETIC_DIRECTIVES.get(lang, PHONETIC_DIRECTIVES['es'])
    header = f"Eres un compositor que piensa nativamente en {lang}. {directive['desc']}\n\n"
    params = f"TEMA: {theme}\nESTILOS: {style_matrix}\nBPM: {bpm}\nTONALIDAD: {key}\n\n"
    rules = [
        "- CERO TRADUCCIÓN literal; piensa en el idioma nativo.",
        "- Marca las sílabas tónicas en MAYÚSCULAS.",
        "- Indica compás y métrica por línea con '|' para cada compás.",
        "- Devuelve: JSON con campos titulo, bpm, tonalidad, estructura, letras (por sección), acordes."
    ]
    # ejemplo de mapeo prosódico
    example_block = ""
    if seed_lines:
        marked = [mark_stress_in_line(line, lang) for line in seed_lines]
        example_block += "EJEMPLOS DE REFERENCIA (con sílabas tónicas marcadas):\n"
        for i, m in enumerate(marked):
            example_block += f"- {m}\n"
            if ipa_enabled:
                ipa = to_ipa(m, lang) or ""
                if ipa:
                    example_block += f"  IPA: {ipa}\n"
    else:
        example_block += "PROPORCIONA 2 LÍNEAS DE REFERENCIA O PIDE 3 VARIANTES.\n"

    # unir todo
    prompt = header + params + "\n".join(rules) + "\n\n" + example_block
    prompt += "\nDEVUELVE SOLO JSON VÁLIDO SIN MARCAJE ADICIONAL.\n"
    return prompt

# -------------------------
# 5. Ejemplo de uso
# -------------------------
if __name__ == "__main__":
    # ejemplo catalán con una línea semilla
    seed = ["La nit és clara i el carrer respira"]
    p = build_phonetic_prompt(language_code="ca", theme="nostàlgia urbana", style_matrix="Indie folk, Rumba catalana", bpm=78, key="Am", seed_lines=seed, ipa_enabled=False)
    print(p)

```python
def validar_cantabilidad_fonetica(letra: str, idioma: str, seccion: str) -> List[str]:
    """
    Valida si la letra es fonéticamente cantable según las reglas del idioma.
    Retorna lista de errores/problemas detectados.
    """
    reglas = PHONETIC_RULES.get(idioma, PHONETIC_RULES["es"])
    errores = []
    
    # 1. Detectar vocales cerradas en secciones de belting (Coro/Estribillo)
    if seccion in ["Coro", "Estribillo", "Chorus"]:
        palabras_agudas = re.findall(r'\b\w*[íúìù]\w*\b', letra, re.IGNORECASE)
        if palabras_agudas and "vocales_cerradas_evitar_agudos" in reglas:
            errores.append(
                f"⚠️ PROSODIA: Detectadas vocales cerradas en '{palabras_agudas}' "
                f"dentro del {seccion}. Fonéticamente difícil de beltear en notas agudas. "
                f"Considera reemplazar con vocales abiertas."
            )
    
    # 2. Detectar vocales nasales en portugués y validar duración
    if idioma == "pt" and "vocales_nasales_largas" in reglas:
        nasales = re.findall(r'\b\w*(?:ão|ãe|em|en)\w*\b', letra, re.IGNORECASE)
        if nasales:
            errores.append(
                f"ℹ️ PROSODIA: Detectadas vocales nasales en '{nasales}'. "
                f"Estas requieren al menos 2 tiempos musicales para resonar correctamente. "
                f"Asegúrate de que la melodía les dé suficiente duración."
            )
    
    # 3. Detectar consonantes percusivas para marcar groove
    if "consonantes_percusivas" in reglas:
        percusivas = []
        for cons in reglas["consonantes_percusivas"]:
            matches = re.findall(rf'\b\w*{cons}\w*\b', letra, re.IGNORECASE)
            percusivas.extend(matches[:3])  # Limitar a 3 ejemplos
        if percusivas:
            errores.append(
                f"✓ PROSODIA: Buen uso de consonantes percusivas: {percusivas[:5]}. "
                f"Estas deben alinearse con los golpes de bombo/caja para maximizar el groove."
            )
    
    # 4. Detectar clusters consonánticos complejos en inglés
    if idioma == "en" and "consonantes_clusters_percusivos" in reglas:
        for cluster in reglas["consonantes_clusters_percusivos"]:
            if cluster.lower() in letra.lower():
                errores.append(
                    f"✓ PROSODIA: Cluster consonántico '{cluster}' detectado. "
                    f"Úsalo como elemento rítmico percusivo en sílabas no acentuadas."
                )
    
    return errores

def generar_prompt_de_correccion(errores: List[str], letra_original: str, idioma: str) -> str:
    """
    Genera un prompt técnico para que la IA corrija los errores prosódicos detectados.
    """
    if not errores:
        return None
    
    prompt = f"""
Eres un experto en prosodia musical y fonética del {idioma}.

La IA ha generado esta letra, pero el Auditor Prosódico ha detectado los siguientes problemas:

{chr(10).join(errores)}

LETRA ORIGINAL:
{letra_original}

INSTRUCCIONES DE CORRECCIÓN:
1. Reescribe la letra manteniendo el significado y la estructura.
2. Aplica las correcciones fonéticas indicadas arriba.
3. Marca las sílabas tónicas en MAYÚSCULAS.
4. Asegúrate de que las vocales abiertas estén en las notas más agudas del {idioma}.
5. Devuelve SOLO la letra corregida, sin explicaciones adicionales.
"""
    return prompt
```

# 2. MOTOR PROSÒDIC (NOU) – Plantilles rítmiques per idioma

def generar_plantilla_prosodica(idioma, num_compases=8):
    """
    Retorna una llista de línies. Cada línia és una llista de símbols:
    'T' = síl·laba tònica (accent fort), 'A' = àtona,
    'O' = vocal oberta (per a belts), 't' = oclusiva sorda preferent,
    'N' = nasal, etc.
    """
    plantilles = {
        "Catalán": {
            "patrones_linea": [
                "A T A T A T",       # ritme iàmbic clàssic
                "T A A T A",         # trocaic
                "A A T A T O",       # final obert per allargar
                "T A T A O",         # accent inicial fort
            ],
            "regles": "Prefereix vocals obertes (a, e, o) en síl·labes tòniques. Sibilants (s, x, ig) en posició àtona per textura."
        },
        "Portugués": {
            "patrones_linea": [
                "A T A T A N",       # final nasal típic
                "T A T A T",         # ginga binària
                "A T A T A A N",     # nasal sostinguda
            ],
            "regles": "Ús abundant de nasals (ão, em). Oclusives (p, t, k) en posició àtona per percussió vocal."
        },
        "Español": {
            "patrones_linea": [
                "A T A T A T",
                "T A A T A",
                "A A T A T",
            ],
            "regles": "Vibrants (rr) en tòniques per actitud. Fricatives (j, ll, y) en àtones."
        },
        "Inglés": {
            "patrones_linea": [
                "T A T A T A",       # stress-timed
                "A T A T A A",
                "T A A T A T",
            ],
            "regles": "Consonant clusters (str, bl, tr) en àtones. Diftongs en tòniques per melismes."
        },
        "Francés": {
            "patrones_linea": [
                "A A T A A T",       # final agut
                "A T A A T A",
                "A A T A T A",
            ],
            "regles": "Liaisons obligatòries. Vocals nasals en síl·labes tòniques. Legato continu."
        }
    }

    datos_idioma = plantilles.get(idioma, plantilles["Español"])
    lineas = []
    for _ in range(num_compases):
        patron = random.choice(datos_idioma["patrones_linea"])
        lineas.append(patron)
    return lineas, datos_idioma["regles"]

MAPA_ACORDES = {
    'C': [60,64,67], 'Cm': [60,63,67], 'Cmaj7': [60,64,67,71], 'Cm7': [60,63,67,70], 'Cm9': [60,63,67,70,74],

    'Am': [57,60,64], 'Am7': [57,60,64,67], 'Am9': [57,60,64,67,71], 'A': [57,61,64], 'A7': [57,61,64,67],
    'Bb': [58,62,65], 'Bbmaj7': [58,62,65,69], 'Bm': [59,62,66], 'B': [59,63,66], 'B7': [59,63,66,69],
    'Eb': [63,67,70], 'Ebmaj7': [63,67,70,74], 'Ab': [68,72,75], 'Db': [61,65,68], 'Dbmaj7': [61,65,68,72]
}

def notas_acorde(ac, fallback="Am"):
    limpio = re.sub(r'[\(\)/].*', '', ac).strip()
    if limpio in MAPA_ACORDES: return MAPA_ACORDES[limpio]
    for k in MAPA_ACORDES:
        if k in limpio: return MAPA_ACORDES[k]
    return MAPA_ACORDES.get(fallback, [57,60,64])

def generar_midi_multipista(cancion, bpm):
    midi = MIDIFile(4)
    tiempo = 0
    for t in range(4): midi.addTempo(t, tiempo, bpm)
    estructura = cancion.get('estructura', [])
    acordes_dict = cancion.get('acordes', {})
    melodia_dict = cancion.get('melodia', {})
    for seccion in estructura:
        acordes_seccion = acordes_dict.get(seccion, [])
        melodia_seccion = melodia_dict.get(seccion, [])
        if not melodia_seccion:
            melodia_seccion = generar_melodia_basica(acordes_seccion)
        for i, ac in enumerate(acordes_seccion):
            notas_ac = notas_acorde(ac)
            for beat in range(4):
                for n in notas_ac:
                    midi.addNote(0, 0, n, tiempo + beat, 0.8, 90)
            root = notas_ac[0] - 12
            fifth = root + 7
            midi.addNote(1, 0, root, tiempo, 1.5, 110)
            midi.addNote(1, 0, fifth, tiempo+2, 1.5, 100)
            for beat in range(4):
                midi.addNote(3, 9, 76 if beat==0 else 77, tiempo+beat, 0.1, 127)
            tiempo += 4
        # Melodia (pista 2)
        tiempo_melodia = 0
        for nota_info in melodia_seccion:
            nota_midi = nombre_a_midi(nota_info['nota'])
            if nota_midi:
                midi.addNote(2, 0, nota_midi, tiempo - (tiempo % 4) + tiempo_melodia, nota_info['duracion'], 100)
            tiempo_melodia += nota_info['duracion']
        tiempo = (tiempo // 4) * 4 + 4
    return midi

def generar_melodia_basica(acordes):
    melodia = []
    for ac in acordes:
        notas = notas_acorde(ac)
        melodia.append({"nota": midi_a_nombre(notas[0]), "duracion": 0.5})
        melodia.append({"nota": midi_a_nombre(notas[1]), "duracion": 0.5})
    return melodia

def nombre_a_midi(nombre):
    mapa = {'C':0,'D':2,'E':4,'F':5,'G':7,'A':9,'B':11}
    match = re.match(r'([A-Ga-g][#b]?)(\d)', nombre)
    if not match: return 60
    nota, octava = match.groups()
    nota = nota.capitalize()
    semitono = mapa.get(nota[0], 0)
    if len(nota) > 1 and nota[1] == '#': semitono += 1
    elif len(nota) > 1 and nota[1] == 'b': semitono -= 1
    return (int(octava) + 1) * 12 + semitono

def midi_a_nombre(num):
    octava = (num // 12) - 1
    nombres = ['C','C#','D','D#','E','F','F#','G','G#','A','A#','B']
    nota = nombres[num % 12]
    return f"{nota}{octava}"

def generar_musicxml(cancion, bpm):
    score = stream.Score()
    part = stream.Part()
    part.append(clef.TrebleClef())
    part.append(meter.TimeSignature('4/4'))
    part.append(tempo.MetronomeMark(number=bpm))
    # Simplificat
    for sec in cancion['estructura']:
        for nota_info in cancion['melodia'].get(sec, []):
            n = note.Note(nota_info['nota'])
            n.duration.quarterLength = nota_info['duracion'] * 4
            part.append(n)
    score.insert(0, part)
    score.metadata = metadata.Metadata()
    score.metadata.title = cancion.get('titulo', 'Obra')
    from music21.musicxml import m21ToXml
    return m21ToXml.GeneralObjectExporter(score).parse().decode('utf-8')

    return texto

Per què aquesta versió sí que funciona?
Plantilla rítmica obligatòria: en lloc de demanar a la IA que «pensi amb fonètica catalana», li donem una seqüència explícita de síl·labes tòniques i àtones, amb indicacions de tipus de so (nasal, obert, oclusiu…). La IA ha d’encaixar les paraules reals en aquest motlle. Així garantim que el ritme de la lletra sigui idiomàticament correcte.

Regles concretes per idioma: cada llengua té les seves pròpies regles (nasals en portuguès, obertes/tancades en català, stress-timing en anglès, etc.) i la plantilla generada les aplica automàticament.

Validació implícita: com que la IA ha de respectar la plantilla, el resultat tindrà una mètrica i una sonoritat molt més properes a la llengua real, no una traducció plana.

### 🚨 Los 4 fallos del código v6.0 que te han pasado:

1. **`music21` es una pesadilla en Streamlit:** Es una librería gigante, lenta y con dependencias complejas. Su función `generar_musicxml` tiene bugs (usa una clase `m21ToXml` que no existe así) y va a crashear.
2. **El validador prosódico es INEXISTENTE:** Le da a la IA una plantilla ("A T A T A T") pero **nunca verifica** si la IA realmente la cumplió. La IA puede ignorarla y el código no se entera.

4. **Mezcla de idiomas caótica:** El prompt está en catalán pero pide el JSON con claves en español (`"titulo"`, `"letras"`). Confunde al modelo.

```python
def contar_silabas(texto, idioma):
    """Cuenta sílabas reales usando diccionario lingüístico, no regex."""
    dic = DICCIONARIOS.get(idioma)
    if not dic: return len(texto.split())
    palabras = re.findall(r'\w+', texto)
    total = 0
    for pal in palabras:
        silabas = dic.inserted(pal).split('-')
        total += len(silabas)
    return total

def detectar_tonicas(texto, idioma):
    """Detecta sílabas tónicas por tildes (ES/CA/PT) o heurística."""
    tildes = {'Español': 'áéíóú', 'Catalán': 'àèéíòóú', 'Portugués': 'áâãàéêíóôõú', 'Francés': 'àâæéèêëîïôœùûü', 'Inglés': ''}
    patron_tildes = tildes.get(idioma, '')
    
    palabras_tonicas = []
    for palabra in re.findall(r'\w+', texto):
        # Si tiene tilde, marcar esa sílaba
        if any(c in palabra.lower() for c in patron_tildes):
            palabras_tonicas.append(palabra.upper())
    return palabras_tonicas

def validar_prosodia(letra, idioma, min_silabas=6, max_silabas=12):
    """Valida si la letra cumple patrones mínimos de cantabilidad."""
    lineas = [l.strip() for l in letra.split('\n') if l.strip()]
    problemas = []
    
    for i, linea in enumerate(lineas):
        silabas = contar_silabas(linea, idioma)
        if silabas < min_silabas:
            problemas.append(f"Línea {i+1} demasiado corta ({silabas} sílabas)")
        elif silabas > max_silabas:
            problemas.append(f"Línea {i+1} demasiado larga ({silabas} sílabas)")
    
    # Detectar falta de tildes en idiomas que las usan
    if idioma in ['Español', 'Catalán', 'Portugués']:
        tonicas = detectar_tonicas(letra, idioma)
        if len(tonicas) < len(lineas) * 0.3:
            problemas.append("Faltan marcas de tonicidad (tildes) - la prosodia será plana")
    
    return problemas

    "Verso 1": ["Am7", "Fmaj7", "Cmaj7", "G7"],
    "Pre-Coro": ["Dm9", "E7alt"],
    "Coro": ["Fmaj7", "Cmaj7", "Am7", "G7"],
    "Verso 2": ["Am7", "Fmaj7", "Cmaj7", "G7"],
    "Puente": ["Fm", "Cm", "Dbmaj7"]
  }},
  "direccion_vocal": "Indicaciones técnicas de interpretación en {idioma}",
  "analisis_prosodico": "Breve explicación de cómo has aplicado las reglas prosódicas del idioma"
}}
"""

def limpiar_json(texto):
    match = re.search(r'\{.*\}', texto, re.DOTALL)
    if match: texto = match.group(0)
    texto = texto.replace("'", '"')
    texto = re.sub(r',\s*([\]}])', r'\1', texto)
    return texto

# ==========================================
# 5. GENERADOR MIDI (simple y robusto)
# ==========================================
MAPA_ACORDES = {
    'C': [60,64,67], 'Cm': [60,63,67], 'Cmaj7': [60,64,67,71], 'Cm7': [60,63,67,70],
    'Dm': [62,65,69], 'Dm7': [62,65,69,72], 'Dm9': [62,65,69,72,76],
    'Em': [64,67,71], 'Em7': [64,67,71,74], 'F': [65,69,72], 'Fmaj7': [65,69,72,76],
    'G': [67,71,74], 'G7': [67,71,74,77], 'Am': [57,60,64], 'Am7': [57,60,64,67],
    'Am9': [57,60,64,67,71], 'Bb': [58,62,65], 'Eb': [63,67,70], 'Db': [61,65,68]
}

def nota_acorde(ac, fallback="Am"):
    limpio = re.sub(r'[\(\)/].*', '', ac).strip()
    if limpio in MAPA_ACORDES: return MAPA_ACORDES[limpio]
    for k in MAPA_ACORDES:
        if k in limpio: return MAPA_ACORDES[k]
    return MAPA_ACORDES.get(fallback, [57,60,64])

def generar_midi(cancion, bpm):
    midi = MIDIFile(2)
    for t in [0,1]: midi.addTempo(t, 0, bpm)
    tiempo = 0
    for sec in cancion.get('estructura', []):
        for ac in cancion.get('acordes', {}).get(sec, []):
            notas = nota_acorde(ac)
            for beat in range(4):
                for n in notas:
                    midi.addNote(0, 0, n, tiempo+beat, 0.8, 90)
            midi.addNote(1, 0, notas[0]-12, tiempo, 1.5, 110)
            midi.addNote(1, 0, notas[0]-5, tiempo+2, 1.5, 100)
            tiempo += 4
    return midi

# ==========================================
# 6. INTERFAZ
# ==========================================
st.title("🎼 Agente Omega v7.0 | Prosodia Validada")
st.caption("Motor que VALIDA la fonética real usando silabación lingüística (pyphen)")

with st.sidebar:

with col1:

with col2:
    if st.button("🗑️ Limpiar"):
        st.session_state.cancion = None
        st.rerun()

if generar and tema:
    intentos = 0
    max_intentos = 2
    cancion_valida = None

                )
                
                limpio = limpiar_json(resp['message']['content'])

                # VALIDACIÓN PROSÓDICA REAL

                problemas_totales = []
                for sec, letra in cancion.get('letras', {}).items():
                    problemas = validar_prosodia(letra, idioma)
                    if problemas:
                        problemas_totales.extend([f"[{sec}] {p}" for p in problemas])
                
                if len(problemas_totales) <= 2:  # Tolerancia de 2 problemas menores
                    cancion_valida = cancion
                    st.success(f"✅ Prosodia validada en intento {intentos}")
                    break
                else:

                    with st.expander("Ver problemas detectados"):

    else:
```

## 🚨 ERRORES CRÍTICOS IDENTIFICADOS Y CORREGIDOS

### Error #1: La IA es "Ciega y Sorda"
**Problema:** Los LLMs no entienden fonética real, solo patrones textuales. Decir "piensa en catalán" no garantiza prosodia correcta.

**Solución:** Implementar **Auditor Prosódico con pyphen** que valida sílabas reales usando diccionarios lingüísticos, no regex.

### Error #2: JSON Malformado
**Problema:** Las IAs devuelven JSON envuelto en markdown o con comas traicioneras.

**Solución:** Función `limpiar_json()` con regex que extrae solo el bloque JSON válido.

### Error #3: Acordes Inventados
**Problema:** La IA genera acordes como "Cmaj13#11b9" que no existen en el mapa.

**Solución:** Función `nota_acorde()` con fallback inteligente que busca coincidencias parciales.

### Error #4: Pérdida de Estado en Streamlit
**Problema:** Al mover sliders, la canción generada desaparecía.

**Solución:** Uso de `st.session_state` para persistencia.

### Error #5: Prosodia Plana
**Problema:** Las letras no respetaban patrones rítmicos del idioma.

**Solución:** **Plantillas Prosódicas Obligatorias** + validación en bucle con reintentos automáticos.

### Error #6: music21 Incompatible
**Problema:** Librería pesada con bugs en Streamlit.

**Solución:** Eliminada, usar solo MIDI con `midiutil`.

## 🌍 REGLAS FONÉTICAS COMPLETAS POR IDIOMA

### 🇪🇸 ESPAÑOL
```
ESTRUCTURA MÉTRICA:
- Versos octosílabos (8 sílabas) o endecasílabos (11 sílabas)
- Acentuación marcada con tildes (á, é, í, ó, ú)
- Sinalefas obligatorias (unir vocales entre palabras)

FONÉTICA:
- Vibrante múltiple (rr) en palabras clave para actitud: "peRRo", "caRRo"
- Fricativas (j, ll, y) en posiciones átonas
- Vocales abiertas (a, e, o) en notas altas para belting
- Consonantes percusivas (p, t, k, ch) alineadas con bombo/caja

EJEMPLO VÁLIDO:
"la CIUdad duERME | pero YO no"
(4-3 | 2-1-1 = 11 sílabas, acentos en posiciones 3, 6, 9)
```

### 🇧🇷 PORTUGUÉS (BRASILEIRO)
```
ESTRUCTURA MÉTRICA:
- Versos con 2-3 vocales nasales por estrofa
- Síncope rítmica (ginga) - palabras cortas entre acentos
- Padrões de 6-8 sílabas con acento en penúltima

FONÉTICA:
- Vocales nasales obligatorias: ão, ãe, em, en
  * "coração" (ko-ɾa-'sɐ̃w̃) - requiere 2 tiempos musicales
  * "mão" (mɐ̃w̃) - nasal sostenida
- Oclusivas (p, t, k) al inicio de verso para percusión
- Vocales abiertas (á, â, é, ê, ó, ô) en belts
- Consonantes suaves en legato

EJEMPLO VÁLIDO:
"A NOite QUENte | na RUA vaZIA"
(2-2 | 1-2-1-2 = 8 sílabas, nasales en "noite" y "vazia")
```

### 🏴 CATALÁN
```
ESTRUCTURA MÉTRICA:
- Versos de 8-10 sílabas (octosíl·labs o decasíl·labs)
- Tildes: à, è, é, í, ò, ó, ú
- Sinalefas complejas entre vocales

FONÉTICA:
- Vocales obertes (a, e, è, o, ò) en posicions tòniques
- Vocals tancades (i, u) evitar en notes agudes
- Sibilants per textura fosca: s, ss, x, ç
- R lleu vibrada en finals de síl·laba
- So de "ig" final (típico catalán): "mig", "raig"

EJEMPLO VÁLIDO:
"La NIT és CLAmma | i el caRRER resPIra"
(1-1-1-2 | 1-1-2-1-2 = 10 sílabas, sibilants en "nit", "carrer")
```

### 🇬🇧 INGLÉS
```
ESTRUCTURA MÉTRICA:
- Stress-timed rhythm: 4 acentos fuertes por verso
- No contar sílabas, contar STRESSES
- Patrones iámbicos (da-DUM da-DUM)

FONÉTICA:
- Consonant clusters como percusión: str-, bl-, tr-, -mp, -nk
  * "STRAY", "BLEED", "DARK", "JUMP"
- Diphthongs en notas largas: ou, ai, ea, ow
  * "NOW", "TIME", "BREAK", "FLOW"
- Schwa (ə) en sílabas átonas
- Glottal stops en estilo urbano

EJEMPLO VÁLIDO:
"STRAY lights BLEED aCROSS the SKY"
(4 stresses: STRAY, BLEED, CROSS, SKY)
```

### 🇫🇷 FRANCÉS
```
ESTRUCTURA MÉTRICA:
- Versos alejandrinos (12 sílabas) o octosílabos (8)
- Liaisons obligatorias entre palabras
- Acento siempre en última sílaba del grupo rítmico

FONÉTICA:
- Vocales nasales en posiciones tónicas: an, in, on, un
  * "enfant" (ɑ̃.fɑ̃), "vin" (vɛ̃), "bon" (bɔ̃)
- Liaisons: "les anges" = le-zɑ̃ʒ
- Legato continuo, sin cortes entre palabras
- R gutural francesa en posiciones medias
- Vocales abiertas (a, e, o) para belting elegante

EJEMPLO VÁLIDO:
"Les ANges PLEUrent | SUR la VILLE"
(3-2 | 1-1-2 = 9 sílabas, nasales en "anges", liaison en "les")
```

```python
import streamlit as st
import ollama
import json
import re
import io
import pyphen
from midiutil import MIDIFile

# ==========================================
# 1. CONFIGURACIÓN Y ESTADO PERSISTENTE
# ==========================================
st.set_page_config(page_title="Omega v7.0 Prosodia Real", page_icon="🎼", layout="wide")

if 'cancion' not in st.session_state:
    st.session_state.cancion = None
if 'midi_buf' not in st.session_state:
    st.session_state.midi_buf = None

# Diccionarios de silabación real por idioma
DICCIONARIOS = {
    "Catalán": pyphen.Pyphen(lang='ca'),
    "Portugués": pyphen.Pyphen(lang='pt_BR'),
    "Español": pyphen.Pyphen(lang='es'),
    "Inglés": pyphen.Pyphen(lang='en_US'),
    "Francés": pyphen.Pyphen(lang='fr')
}

# ==========================================
# 2. MOTOR DE VALIDACIÓN PROSÓDICA
# ==========================================
def contar_silabas(texto, idioma):
    """Cuenta sílabas reales usando diccionario lingüístico."""
    dic = DICCIONARIOS.get(idioma)
    if not dic: return len(texto.split())
    palabras = re.findall(r'\w+', texto)
    total = 0
    for pal in palabras:
        silabas = dic.inserted(pal).split('-')
        total += len(silabas)
    return total

def detectar_tonicas(texto, idioma):
    """Detecta sílabas tónicas por tildes."""
    tildes = {
        'Español': 'áéíóú',
        'Catalán': 'àèéíòóú',
        'Portugués': 'áâãàéêíóôõú',
        'Francés': 'àâæéèêëîïôœùûü',
        'Inglés': ''
    }
    patron_tildes = tildes.get(idioma, '')
    palabras_tonicas = []
    for palabra in re.findall(r'\w+', texto):
        if any(c in palabra.lower() for c in patron_tildes):
            palabras_tonicas.append(palabra.upper())
    return palabras_tonicas

def validar_prosodia(letra, idioma, min_silabas=6, max_silabas=12):
    """Valida patrones de cantabilidad."""
    lineas = [l.strip() for l in letra.split('\n') if l.strip()]
    problemas = []
    
    for i, linea in enumerate(lineas):
        silabas = contar_silabas(linea, idioma)
        if silabas < min_silabas:
            problemas.append(f"Línea {i+1} demasiado corta ({silabas} sílabas)")
        elif silabas > max_silabas:
            problemas.append(f"Línea {i+1} demasiado larga ({silabas} sílabas)")
    
    if idioma in ['Español', 'Catalán', 'Portugués']:
        tonicas = detectar_tonicas(letra, idioma)
        if len(tonicas) < len(lineas) * 0.3:
            problemas.append("Faltan marcas de tonicidad")
    
    return problemas

# ==========================================
# 3. PLANTILLAS PROSÓDICAS POR IDIOMA
# ==========================================
PLANTILLAS = {
    "Catalán": """
- Versos de 8-10 sílabas (octosíl·labs o decasíl·labs)
- Ús de vocals obertes (a, è, o) en posicions tòniques
- Sibilants (s, ss, x, ç) per textura fosca
- Exemples: "La NIT és CLAmma i el caRRER resPIra"
""",
    "Portugués": """
- Versos con 2-3 vocales nasales por estrofa (ão, ãe, em)
- Síncope rítmica (ginga) - palabras cortas entre acentos
- Oclusivas (p, t, k) al inicio de verso para percusión
- Ejemplo: "A NOite QUENte | na RUA vaZIA"
""",
    "Español": """
- Versos octosílabos o endecasílabos
- Vibrante múltiple (rr) en palabras clave para actitud
- Tilde prosódica marcada en sílabas fuertes
- Ejemplo: "la CIUdad duERME | pero YO no"
""",
    "Inglés": """
- Stress-timed rhythm: 4 acentos fuertes por verso
- Consonant clusters (str-, bl-, -mp) como percusión
- Diphthongs (ou, ai, ea) en notas largas
- Example: "STRAY lights BLEED aCROSS the SKY"
""",
    "Francés": """
- Versos alejandrinos (12 sílabas) o octosílabos
- Liaisons obligatorias entre palabras
- Vocales nasales (an, in, on, un) al final de frase
- Exemple: "Les ANges PLEUrent SUR la VILLE"
"""
}

# ==========================================
# 4. CONSTRUCTOR DE PROMPT MAESTRO
# ==========================================
def construir_prompt(idioma, estilos, tema, bpm, tonalidad):
    plantilla = PLANTILLAS.get(idioma, PLANTILLAS["Español"])
    
    return f"""Eres el AGENTE OMEGA v7.0, compositor experto en prosodia.

🌍 IDIOMA NATIVO: {idioma} (ESCRIBE DIRECTAMENTE EN ESTE IDIOMA, CERO TRADUCCIÓN)

📏 REGLAS PROSÓDICAS OBLIGATORIAS PARA {idioma}:
{plantilla}

⚠️ VALIDACIÓN AUTOMÁTICA:
Tu salida será validada por un módulo lingüístico que:
1. Contará sílabas reales por línea (deben ser 6-12)
2. Detectará tildes y marcas de tonicidad
3. Verificará la presencia de rasgos fonéticos del idioma
Si no cumples, se te pedirá reescribir.

🎵 PARÁMETROS:
- Tema: {tema}
- BPM: {bpm} | Tonalidad: {tonalidad}
- Estilos: {estilos}

🎭 FUSIÓN DE ESTILOS:
Aplica las matrices de: Lady Gaga (teatralidad), The Smiths (melancolía modal), 
Anitta (groove percusivo), Nathy Peluso (tensión armónica).

📤 DEVUELVE JSON VÁLIDO con esta estructura EXACTA:
{{
  "titulo": "...",
  "estructura": ["Verso 1", "Pre-Coro", "Coro", "Verso 2", "Puente", "Coro"],
  "letras": {{
    "Verso 1": "línea 1\\nlínea 2\\nlínea 3\\nlínea 4",
    "Pre-Coro": "...",
    "Coro": "...",
    "Verso 2": "...",
    "Puente": "..."
  }},
  "acordes": {{
    "Verso 1": ["Am7", "Fmaj7", "Cmaj7", "G7"],
    "Pre-Coro": ["Dm9", "E7alt"],
    "Coro": ["Fmaj7", "Cmaj7", "Am7", "G7"],
    "Verso 2": ["Am7", "Fmaj7", "Cmaj7", "G7"],
    "Puente": ["Fm", "Cm", "Dbmaj7"]
  }},
  "direccion_vocal": "Indicaciones técnicas de interpretación en {idioma}",
  "analisis_prosodico": "Breve explicación de cómo has aplicado las reglas prosódicas del idioma"
}}
"""

def limpiar_json(texto):
    match = re.search(r'\{.*\}', texto, re.DOTALL)
    if match: texto = match.group(0)
    texto = texto.replace("'", '"')
    texto = re.sub(r',\s*([\]}])', r'\1', texto)
    return texto

# ==========================================
# 5. GENERADOR MIDI MULTI-PISTA
# ==========================================
MAPA_ACORDES = {
    'C': [60,64,67], 'Cm': [60,63,67], 'Cmaj7': [60,64,67,71], 'Cm7': [60,63,67,70],
    'Cm9': [60,63,67,70,74], 'Dm': [62,65,69], 'Dm7': [62,65,69,72], 'Dm9': [62,65,69,72,76],
    'D': [62,66,69], 'D7': [62,66,69,72], 'Em': [64,67,71], 'Em7': [64,67,71,74],
    'E': [64,68,71], 'E7': [64,68,71,74], 'F': [65,69,72], 'Fmaj7': [65,69,72,76],
    'Fm': [65,68,72], 'F#m': [66,69,73], 'G': [67,71,74], 'G7': [67,71,74,77],
    'Gmaj7': [67,71,74,78], 'Gm': [67,70,74], 'Am': [57,60,64], 'Am7': [57,60,64,67],
    'Am9': [57,60,64,67,71], 'A': [57,61,64], 'A7': [57,61,64,67],
    'Bb': [58,62,65], 'Bbmaj7': [58,62,65,69], 'Bm': [59,62,66], 'B': [59,63,66],
    'B7': [59,63,66,69], 'Eb': [63,67,70], 'Ebmaj7': [63,67,70,74],
    'Ab': [68,72,75], 'Db': [61,65,68], 'Dbmaj7': [61,65,68,72]
}

def nota_acorde(ac, fallback="Am"):
    limpio = re.sub(r'[\(\)/].*', '', ac).strip()
    if limpio in MAPA_ACORDES: return MAPA_ACORDES[limpio]
    for k in MAPA_ACORDES:
        if k in limpio: return MAPA_ACORDES[k]
    return MAPA_ACORDES.get(fallback, [57,60,64])

def generar_midi(cancion, bpm):
    """Genera MIDI con 3 pistas: Acordes, Bajo, Click"""
    midi = MIDIFile(3)
    for t in [0,1,2]: midi.addTempo(t, 0, bpm)
    tiempo = 0
    
    for sec in cancion.get('estructura', []):
        for ac in cancion.get('acordes', {}).get(sec, []):
            notas = nota_acorde(ac)
            
            # Pista 0: Acordes (4 golpes por compás)
            for beat in range(4):
                for n in notas:
                    midi.addNote(0, 0, n, tiempo+beat, 0.8, 90)
            
            # Pista 1: Bajo (tónica y quinta)
            root = notas[0] - 12
            fifth = notas[2] - 12 if len(notas) > 2 else root + 7
            midi.addNote(1, 0, root, tiempo, 1.5, 110)
            midi.addNote(1, 0, fifth, tiempo+2, 1.5, 100)
            
            # Pista 2: Click (guía rítmica)
            for beat in range(4):
                midi.addNote(2, 9, 76 if beat==0 else 77, tiempo+beat, 0.1, 127)
            
            tiempo += 4
    return midi

    tonalidad = st.selectbox("🎹 Tonalidad", [

    ])

with col1:

with col2:
    if st.button("🗑️ Limpiar"):
        st.session_state.cancion = None
        st.rerun()

# ==========================================
# 7. LÓGICA DE GENERACIÓN CON VALIDACIÓN
# ==========================================
if generar and tema:
    intentos = 0
    max_intentos = 2
    cancion_valida = None
    
    while intentos <= max_intentos:
        intentos += 1
        with st.spinner(f"🧠 Intento {intentos}: Componiendo en {idioma}..."):
            try:
                prompt = construir_prompt(idioma, estilos, tema, bpm, tonalidad)
                resp = ollama.chat(
                    model=modelo,
                    messages=[
                        {'role':'system','content':'Compositor experto. Devuelve SOLO JSON.'},
                        {'role':'user','content':prompt}
                    ],
                    format='json',
                    options={'temperature':0.85, 'num_predict':8192}
                )
                
                limpio = limpiar_json(resp['message']['content'])
                cancion = json.loads(limpio)
                
                # VALIDACIÓN PROSÓDICA REAL
                st.info(f"🔍 Intento {intentos}: Validando prosodia...")
                problemas_totales = []
                for sec, letra in cancion.get('letras', {}).items():
                    problemas = validar_prosodia(letra, idioma)
                    if problemas:
                        problemas_totales.extend([f"[{sec}] {p}" for p in problemas])
                
                if len(problemas_totales) <= 2:
                    cancion_valida = cancion
                    st.success(f"✅ Prosodia validada en intento {intentos}")
                    break
                else:
                    st.warning(f"⚠️ {len(problemas_totales)} problemas. Reintentando...")
                    with st.expander("Ver problemas detectados"):
                        for p in problemas_totales:
                            st.write(f"• {p}")
            except Exception as e:
                st.error(f"Error en intento {intentos}: {e}")
    
    if cancion_valida:
        midi = generar_midi(cancion_valida, bpm)
        buf = io.BytesIO()
        midi.writeFile(buf)
        buf.seek(0)
        st.session_state.cancion = cancion_valida
        st.session_state.midi_buf = buf.getvalue()
        st.rerun()
    else:
        st.error("❌ No se logró generar letra con prosodia válida.")

# ==========================================
# 8. VISUALIZACIÓN DE RESULTADOS
# ==========================================
if st.session_state.cancion:
    c = st.session_state.cancion
    st.markdown("---")
    st.subheader(f"🎵 {c.get('titulo','Sin título')}")
    st.info(f"🗣️ **Dirección vocal:** {c.get('direccion_vocal','')}")
    
    if c.get('analisis_prosodico'):

    col1, col2 = st.columns(2)
    txt = f"Título: {c.get('titulo')}\nIdioma: {idioma} | BPM: {bpm} | Tonalidad: {tonalidad}\n\n"
    for sec in c.get('estructura', []):
        txt += f"[{sec}]\nAcordes: {' | '.join(c.get('acordes',{}).get(sec,[]))}\n"
        txt += f"{c.get('letras',{}).get(sec,'')}\n\n"
    
    col1.download_button("📄 Letra + Acordes (TXT)", txt, 
        f"{c.get('titulo')}.txt", use_container_width=True)
    col2.download_button("🎹 MIDI Multi-Pista", st.session_state.midi_buf, 
        f"{c.get('titulo')}.mid", mime="audio/midi", use_container_width=True)
    
    for sec in c.get('estructura', []):
        with st.expander(f"🎼 {sec}", expanded=(sec=="Coro")):
            st.code(" | ".join(c.get('acordes',{}).get(sec,[])))
            letra = c.get('letras',{}).get(sec,'')
            st.markdown(letra)
            silabas = contar_silabas(letra, idioma)
            st.caption(f"📏 Total sílabas reales: {silabas}")
```

## 📊 ARQUITECTURA DEL SISTEMA

### Flujo de Trabajo
```
Usuario → Parámetros (idioma, estilos, tema, BPM, tonalidad)
    ↓
Constructor de Prompt (inyecta plantillas prosódicas)
    ↓
Ollama (Qwen 2.5 7B) genera JSON con letra + acordes
    ↓
Validador Prosódico (pyphen cuenta sílabas reales)
    ↓
¿Válido? → SÍ: Generar MIDI + Mostrar
         → NO: Reintentar (máx 3 veces)
    ↓
Exportar TXT + MIDI multi-pista
```

### Componentes Críticos
1. **pyphen** - Silabación lingüística real por idioma
2. **Ollama** - Ejecución local de LLM sin internet
3. **midiutil** - Generación de archivos MIDI estándar
4. **Streamlit** - Interfaz web local
5. **Validación en bucle** - Asegura calidad prosódica
